### 5. CSV Generation
**File:** `src/lib/reports/csv.ts`

```typescript
export function generateCsvReport(config: ReportConfig, data: any[]): string {
  // Simple CSV with headers
  // All values properly escaped
  // Date/time formatted for spreadsheet consumption
  const headers = config.columns.map(c => c.label).join(',');
  const rows = data.map(row =>
    config.columns.map(c => escapeCSV(row[c.key])).join(',')
  );
  return [headers, ...rows].join('\n');
}
```

### 6. Excel Generation
**File:** `src/lib/reports/excel.ts`

```typescript
import * as XLSX from 'xlsx';

export function generateExcelReport(config: ReportConfig, data: any[]): Buffer {
  const wb = XLSX.utils.book_new();

  // Main data sheet
  const ws = XLSX.utils.json_to_sheet(data, { header: config.columns.map(c => c.key) });

  // Apply column headers (friendly names)
  config.columns.forEach((col, idx) => {
    const cellRef = XLSX.utils.encode_cell({ r: 0, c: idx });
    ws[cellRef].v = col.label;
  });

  // Column widths
  ws['!cols'] = config.columns.map(c => ({ wch: c.width || 15 }));

  XLSX.utils.book_append_sheet(wb, ws, config.sheetName || 'Report');

  // Summary sheet (optional)
  if (config.includeSummary) {
    const summaryWs = XLSX.utils.json_to_sheet(config.summaryData);
    XLSX.utils.book_append_sheet(wb, summaryWs, 'Summary');
  }

  return Buffer.from(XLSX.write(wb, { type: 'buffer', bookType: 'xlsx' }));
}
```

### 8. Scheduled Reports
**Implementation:** Vercel Cron Job or Supabase Edge Function

**API Route:** `src/app/api/scheduled-reports/route.ts`

**Daily Summary (runs every morning at 7am facility time):**
- Covers previous day
- Includes: checklist completion %, ice makes count, any incidents, any out-of-range alerts
- Sent as email to configured recipients

**Weekly Summary (runs Monday morning at 7am):**
- Covers previous 7 days
- Includes: all daily summary data aggregated, scheduling hours summary, trend indicators
- Sent as email with PDF attachment

**Admin Configuration:**
- In Admin Control Center → add "Scheduled Reports" section
- Toggle daily/weekly reports on/off
- Configure recipient list per report type
- Set preferred delivery time

**Tables needed (add migration if not present):**
```sql
CREATE TABLE scheduled_report_settings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  facility_id UUID NOT NULL REFERENCES facilities(id) ON DELETE CASCADE,
  report_type TEXT NOT NULL, -- 'daily_summary', 'weekly_summary'
  is_enabled BOOLEAN DEFAULT FALSE,
  recipients UUID[] DEFAULT '{}', -- array of profile IDs
  delivery_hour INTEGER DEFAULT 7, -- hour in facility timezone
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE(facility_id, report_type)
);
```

### 9. Download Flow
```
User clicks "Generate" → Loading spinner
  → Server action queries data, generates file
  → File uploaded to Supabase Storage (bucket: 'reports', path: 'facility_id/timestamp_reportname.pdf')
  → Signed URL returned (24hr expiry)
  → Browser triggers download from signed URL
  → Toast: "Report downloaded successfully"
```
